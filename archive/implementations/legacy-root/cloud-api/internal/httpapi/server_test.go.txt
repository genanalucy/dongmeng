type testAPI struct{ handler http.Handler; store *store.Memory; tokens auth.TokenIssuer; now time.Time }
func newTestAPI()testAPI{m:=store.NewMemory();tokens:=auth.TokenIssuer{Issuer:"test",Audience:"clients",AccessSecret:bytes.Repeat([]byte("a"),32),SessionSecret:bytes.Repeat([]byte("s"),32)};return testAPI{handler:New(m,tokens,slog.New(slog.NewTextHandler(io.Discard,nil)),[]string{"https://app.example"},"test",1<<20),store:m,tokens:tokens,now:time.Now()}}
func (a testAPI) register(t *testing.T,email string,now time.Time)domain.User{t.Helper();hash,err:=auth.HashPassword("correct horse battery");if err!=nil{t.Fatal(err)};u,_,err:=a.store.Register(t.Context(),domain.RegisterParams{Email:email,PasswordHash:hash,Now:now});if err!=nil{t.Fatal(err)};return u}
func (a testAPI) token(t *testing.T,u domain.User)string{t.Helper();v,err:=a.tokens.AccessToken(u.ID,u.Role,time.Hour,time.Now());if err!=nil{t.Fatal(err)};return v}
func request(t *testing.T,h http.Handler,method,path,token string,body any)*httptest.ResponseRecorder{t.Helper();var r io.Reader;if body!=nil{raw,err:=json.Marshal(body);if err!=nil{t.Fatal(err)};r=bytes.NewReader(raw)};req:=httptest.NewRequest(method,path,r);if body!=nil{req.Header.Set("Content-Type","application/json")};if token!=""{req.Header.Set("Authorization","Bearer "+token)};rr:=httptest.NewRecorder();h.ServeHTTP(rr,req);return rr}

func TestUnknownOriginRejectedAndRequestIDReturned(t *testing.T){a:=newTestAPI();req:=httptest.NewRequest(http.MethodGet,"/healthz",nil);req.Header.Set("Origin","https://evil.example");rr:=httptest.NewRecorder();a.handler.ServeHTTP(rr,req);if rr.Code!=http.StatusForbidden{t.Fatalf("status=%d",rr.Code)};if rr.Header().Get("X-Request-ID")==""{t.Fatal("missing request id response header")}}

func TestTranslationTokenClaims(t *testing.T){a:=newTestAPI();u:=a.register(t,"session@example.com",time.Now());rr:=request(t,a.handler,http.MethodPost,"/v1/translation-sessions",a.token(t,u),map[string]any{});if rr.Code!=http.StatusCreated{t.Fatalf("status=%d body=%s",rr.Code,rr.Body.String())};var out struct{SessionID uuid.UUID `json:"session_id"`;Token string `json:"token"`;Scope string `json:"scope"`};if err:=json.Unmarshal(rr.Body.Bytes(),&out);err!=nil{t.Fatal(err)};if out.SessionID==uuid.Nil||out.Token==""||out.Scope!="translation"{t.Fatalf("bad response: %+v",out)}}
