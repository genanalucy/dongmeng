package config

import (
	"fmt"
	"os"
	"strconv"
	"strings"
	"time"
)

type Config struct {
	AppEnv, Address, DatabaseURL, JWTIssuer, JWTAudience string
	JWTAccessSecret, JWTSessionSecret                    []byte
	AllowedOrigins                                       []string
	AccessTTL, RefreshTTL, SessionTTL                    time.Duration
	BodyLimit                                            int64
}

func Load() (Config, error) {
	cfg := Config{
		AppEnv: env("APP_ENV", "production"), Address: env("HTTP_ADDRESS", ":8080"),
		DatabaseURL: os.Getenv("DATABASE_URL"), JWTIssuer: env("JWT_ISSUER", "dngmeng-cloud-api"),
		JWTAudience: env("JWT_AUDIENCE", "dngmeng-clients"), AllowedOrigins: splitCSV(os.Getenv("CORS_ALLOWED_ORIGINS")),
		AccessTTL: 15 * time.Minute, RefreshTTL: 30 * 24 * time.Hour, SessionTTL: 5 * time.Minute, BodyLimit: 1 << 20,
		JWTAccessSecret: []byte(os.Getenv("JWT_ACCESS_SECRET")), JWTSessionSecret: []byte(os.Getenv("JWT_SESSION_SECRET")),
	}
	if v := os.Getenv("HTTP_BODY_LIMIT_BYTES"); v != "" {
		n, err := strconv.ParseInt(v, 10, 64)
		if err != nil {
			return Config{}, fmt.Errorf("HTTP_BODY_LIMIT_BYTES: %w", err)
		}
		cfg.BodyLimit = n
	}
	if cfg.DatabaseURL == "" {
		return Config{}, fmt.Errorf("DATABASE_URL is required")
	}
	if len(cfg.JWTAccessSecret) < 32 || len(cfg.JWTSessionSecret) < 32 {
		return Config{}, fmt.Errorf("JWT secrets must each be at least 32 bytes")
	}
	if len(cfg.AllowedOrigins) == 0 {
		return Config{}, fmt.Errorf("CORS_ALLOWED_ORIGINS requires an exact allowlist")
	}
	for _, origin := range cfg.AllowedOrigins {
		if origin == "*" {
			return Config{}, fmt.Errorf("wildcard CORS origin is forbidden")
		}
	}
	return cfg, nil
}

func env(name, fallback string) string {
	if v := os.Getenv(name); v != "" {
		return v
	}
	return fallback
}
func splitCSV(value string) []string {
	var out []string
	for _, v := range strings.Split(value, ",") {
		if v = strings.TrimSpace(v); v != "" {
			out = append(out, v)
		}
	}
	return out
}
